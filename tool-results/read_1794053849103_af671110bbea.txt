    37→    37→/**
    38→    38→ * after a successful mutation: invalidate caches on this device.
    39→    39→ * (Other devices are updated by the server-side broadcast in the API route.)
    40→    40→ */
    41→    41→function useSyncedMutation<TVars, TData>(
    42→    42→  fn: (vars: TVars) => Promise<TData>,
    43→    43→  opts: {
    44→    44→    invalidate?: readonly (readonly string[])[]
    45→    45→    success?: string
    46→    46→  } = {}
    47→    47→) {
    48→    48→  const queryClient = useQueryClient()
    49→    49→  return useMutation({
    50→    50→    mutationFn: fn,
    51→    51→    onSuccess: () => {
    52→    52→      for (const key of opts.invalidate ?? []) queryClient.invalidateQueries({ queryKey: key })
    53→    53→      if (opts.success) toast({ description: opts.success })
    54→    54→    },
    55→    55→    onError: (error) => {
    56→    56→      if (error instanceof OfflineQueuedError) {
    57→    57→        toast({ description: error.message })
    58→    58→        return
    59→    59→      }
    60→    60→      toast({
    61→    61→        variant: 'destructive',
    62→    62→        description: error instanceof Error ? error.message : 'Something went wrong.',
    63→    63→      })
    64→    64→    },
    65→    65→  })
    66→    66→}
