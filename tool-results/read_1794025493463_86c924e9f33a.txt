    36→/**
    37→ * after a successful mutation: invalidate caches on this device.
    38→ * (Other devices are updated by the server-side broadcast in the API route.)
    39→ */
    40→function useSyncedMutation<TVars, TData>(
    41→  fn: (vars: TVars) => Promise<TData>,
    42→  opts: {
    43→    invalidate?: string[][]
    44→    success?: string
    45→  } = {}
    46→) {
    47→  const queryClient = useQueryClient()
    48→  return useMutation({
    49→    mutationFn: fn,
    50→    onSuccess: () => {
    51→      for (const key of opts.invalidate ?? []) queryClient.invalidateQueries({ queryKey: key })
    52→      if (opts.success) toast({ description: opts.success })
    53→    },
    54→    onError: (error) => {
    55→      if (error instanceof OfflineQueuedError) {
    56→        toast({ description: error.message })
    57→        return
    58→      }
    59→      toast({
    60→        variant: 'destructive',
    61→        description: error instanceof Error ? error.message : 'Something went wrong.',
    62→      })
    63→    },
    64→  })
    65→}
