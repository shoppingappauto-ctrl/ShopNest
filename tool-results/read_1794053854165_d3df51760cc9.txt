    37→/**
    38→ * after a successful mutation: invalidate caches on this device.
    39→ * (Other devices are updated by the server-side broadcast in the API route.)
    40→ */
    41→function useSyncedMutation<TVars, TData>(
    42→  fn: (vars: TVars) => Promise<TData>,
    43→  opts: {
    44→    invalidate?: readonly (readonly string[])[]
    45→    success?: string
    46→  } = {}
    47→) {
    48→  const queryClient = useQueryClient()
    49→  return useMutation({
    50→    mutationFn: fn,
    51→    onSuccess: () => {
    52→      for (const key of opts.invalidate ?? []) queryClient.invalidateQueries({ queryKey: key })
    53→      if (opts.success) toast({ description: opts.success })
    54→    },
    55→    onError: (error) => {
    56→      if (error instanceof OfflineQueuedError) {
    57→        toast({ description: error.message })
    58→        return
    59→      }
    60→      toast({
    61→        variant: 'destructive',
    62→        description: error instanceof Error ? error.message : 'Something went wrong.',
    63→      })
    64→    },
    65→  })
    66→}
