    36→    36→/**
    37→    37→ * after a successful mutation: invalidate caches on this device.
    38→    38→ * (Other devices are updated by the server-side broadcast in the API route.)
    39→    39→ */
    40→    40→function useSyncedMutation<TVars, TData>(
    41→    41→  fn: (vars: TVars) => Promise<TData>,
    42→    42→  opts: {
    43→    43→    invalidate?: string[][]
    44→    44→    success?: string
    45→    45→  } = {}
    46→    46→) {
    47→    47→  const queryClient = useQueryClient()
    48→    48→  return useMutation({
    49→    49→    mutationFn: fn,
    50→    50→    onSuccess: () => {
    51→    51→      for (const key of opts.invalidate ?? []) queryClient.invalidateQueries({ queryKey: key })
    52→    52→      if (opts.success) toast({ description: opts.success })
    53→    53→    },
    54→    54→    onError: (error) => {
    55→    55→      if (error instanceof OfflineQueuedError) {
    56→    56→        toast({ description: error.message })
    57→    57→        return
    58→    58→      }
    59→    59→      toast({
    60→    60→        variant: 'destructive',
    61→    61→        description: error instanceof Error ? error.message : 'Something went wrong.',
    62→    62→      })
    63→    63→    },
    64→    64→  })
    65→    65→}
